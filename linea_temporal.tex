\subsection*{5. Método Taylor de Erosión -- Borrador (Mar 2026)}
Método alternativo de detección de erosión propuesto para reemplazar el filtro $z > z_{\text{max}} \wedge v_z > 0$ del método v1.

\subsubsection*{Descripción del método}
\begin{enumerate}
    \item \textbf{Selección de lonja}: partículas con $z > z_{\text{max}} - 10R$ (sin filtro previo de $v_z$ ni de tipo)
    \item \textbf{Cluster analysis}: ejecutar OVITO sobre toda la lonja (coordinación + cluster)
    \item \textbf{Filtro por centro de masa del cluster}:
        \begin{itemize}
            \item $v_{z,\text{cm}} > v_{z,\text{min}}$ (el CM se mueve hacia arriba)
            \item $z_{\text{cm}} > z_{\text{max}}$ (el CM está por encima de la superficie)
            \item $|\Sigma f_z| < \varepsilon$ (fuerza neta $\approx 0$, vuelo balístico)
        \end{itemize}
    \item \textbf{Filtro individual}: partículas con $z > z_{\text{max}}$ y $v_z > 0$
    \item Incluir siempre proyectil (tipo 2)
\end{enumerate}

\subsubsection*{Archivos modificados}
\begin{itemize}
    \item \texttt{analysis/topology.py}: \texttt{analyze\_erosion\_taylor()}, \texttt{\_filter\_ballistic\_clusters\_taylor()}, \texttt{run\_topology\_analysis()} con parámetro \texttt{method}
    \item \texttt{config.py}: \texttt{FilterConfig.lonja\_radios: int} + parsing en \texttt{\_build\_config()}
    \item \texttt{config.toml}: \texttt{lonja\_radios = 10} en sección \texttt{[filtros]}
    \item \texttt{io/result_writer.py}: parámetro \texttt{suffix} opcional en \texttt{save\_erosion\_results()}
    \item \texttt{plot.py}: parámetro \texttt{suffix} opcional en \texttt{plot\_erosion()}
    \item \texttt{pipeline.py}: wiring con \texttt{method="taylor"}, \texttt{suffix="\_filtrado\_taylor"}, pasa \texttt{radio\_m}, \texttt{lonja\_radios}, \texttt{umbral\_vz\_min}
\end{itemize}

\subsubsection*{Intentos fallidos}
\begin{enumerate}
    \item \textbf{Intento 1} (commit \texttt{a2b9265}): lonja $z > z_{\text{max}} - 10R$, filtros CM del cluster, filtro individual $z > z_{\text{max}}$ \\
        \textit{Resultado: 25 partículas (esperado: 20). Se devolvían partículas del cluster que estaban por debajo de $z_{\text{max}}$ antes de agregar el filtro individual.}
    \item \textbf{Intento 2} (commit \texttt{9aef9ab} + reset): se agregó filtro $z > z_{\text{max}}$ individual \\
        \textit{Resultado: 25 partículas (sin cambio).}
    \item \textbf{Intento 3} (intentado localmente): se verificó que la lonja no era el problema (el filtro CM debería descartar clusters gigantes de la cama) \\
        \textit{Hipótesis descartada: clusters de la cama pasan filtros CM.}
    \item \textbf{Intento 4} (commit \texttt{05eb217}): se agregó $v_z > 0$ al filtro individual de partículas \\
        \textit{Resultado: 25 partículas (sin cambio).}
\end{enumerate}

\subsubsection*{Estado}
\begin{itemize}
    \item Código original (método v1) preservado en rama \texttt{desarrollo} --- funciona correctamente (20 partículas)
    \item Historial completo de intentos en rama \texttt{draft/taylor-erosion}
    \item \textbf{Causa del sobreconteo no identificada}: posible diferencia en cómo se acumulan partículas a lo largo de los timesteps (v1 persiste flag \texttt{eroded}, taylor re-evalúa clusters cada frame)
\end{itemize}
